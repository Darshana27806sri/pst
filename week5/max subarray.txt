import java.io.BufferedReader;
import java.io.InputStreamReader;
import java.io.IOException;
import java.util.StringTokenizer;
public class Solution {
    public static void main(String[] args) throws IOException {
        BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
        String line = br.readLine();
        if (line == null || line.trim().isEmpty()) {
            return;
        }
        int t = Integer.parseInt(line.trim());
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < t; i++) {
            String nLine = br.readLine();
            if (nLine == null) break;
            int n = Integer.parseInt(nLine.trim());
            String arrLine = br.readLine();
            if (arrLine == null) break;
            StringTokenizer st = new StringTokenizer(arrLine);
            int[] arr = new int[n];
            for (int j = 0; j < n; j++) {
                arr[j] = Integer.parseInt(st.nextToken());
            }
            sb.append(solveMaxSubarray(arr)).append("\n");
        }
        System.out.print(sb.toString());
    }
    private static String solveMaxSubarray(int[] arr) {
        int maxSubarraySum = arr[0];
        int currentSubarraySum = arr[0];
        int maxSubsequenceSum = arr[0];
        int maxElement = arr[0];
        for (int i = 1; i < arr.length; i++) {
            currentSubarraySum = Math.max(arr[i], currentSubarraySum + arr[i]);
            maxSubarraySum = Math.max(maxSubarraySum, currentSubarraySum);
            if (arr[i] > maxElement) {
                maxElement = arr[i];
            }
        }
        if (maxElement < 0) {
            maxSubsequenceSum = maxElement;
        } else {
            maxSubsequenceSum = 0;
            for (int val : arr) {
                if (val > 0) {
                    maxSubsequenceSum += val;
                }
            }
        }
        return maxSubarraySum + " " + maxSubsequenceSum;
    }
}
