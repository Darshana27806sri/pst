import java.util.*;
class Student{}
class Rockstar{}
class Hacker{}
public class Solution{
public static void main(String[] args){
Scanner sc=new Scanner(System.in);
int n=sc.nextInt();
Object[] objects=new Object[n];
for(int i=0;i<n;i++){
int type=sc.nextInt();
if(type==1) objects[i]=new Student();
else if(type==2) objects[i]=new Rockstar();
else objects[i]=new Hacker();
}
int student=0,rockstar=0,hacker=0;
for(Object obj:objects){
if(obj instanceof Student) student++;
if(obj instanceof Rockstar) rockstar++;
if(obj instanceof Hacker) hacker++;
}
System.out.println(student+" "+rockstar+" "+hacker);
}
}
